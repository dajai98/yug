===== FILE: training\corpus_prep.py =====
"""Corpus Preparation Prototype.

VERIFICATION STATUS: this module itself has no PyTorch dependency (only
``dataclasses``, ``enum``, ``hashlib``, ``json``, ``re`` -- stdlib) and
was executed directly in this environment (see the accompanying test
file's own VERIFICATION STATUS note for exact results). Only the one
test proving compatibility with ``training.dataset``'s real API needs
PyTorch (that module imports ``torch`` at module scope) and is gated
with ``pytest.importorskip("torch")``, unexecuted here, matching every
other torch-dependent test already in this repository.

Scope (deliberately small -- this is a correctness prototype, not the
future large-scale pipeline; see this module's own docstring sections
below for exactly what is and is not implemented):

    Raw documents (RawDocument)
          |
          v
    License gate (ALLOW / REJECT / REVIEW, explicit policy only)
          |
          v
    Normalization (conservative, deterministic)
          |
          v
    Quality filter (empty / below-min-length / pathological repetition)
          |
          v
    Exact deduplication (SHA-256 of normalized text, first-seen wins)
          |
          v
    Deterministic train/validation split (reuses training.split.split_sentences)
          |
          v
    AcceptedDocument list + CorpusManifest
          |
          v
    (caller feeds AcceptedDocument.text into training.dataset.encode_corpus,
     unmodified -- see this module's docstring section on integration)

Explicitly NOT implemented (out of scope for this prototype, per its
own directive's hard boundaries): ML quality/toxicity/perplexity
classifiers, near-duplicate detection (MinHash/LSH/embeddings),
web-scale ingestion, a dataset UI, external services. All rejection
decisions are simple, deterministic, and explainable from the input
text and an explicitly configured license policy alone.

Architectural boundary: this module lives under ``training/`` and may
import ``yug`` (it doesn't need to -- it has no dependency on
``yug.native``/``yug.tokenizer`` at all). Nothing under ``yug/``
imports this module or anything else under ``training/``.
"""

from __future__ import annotations

import hashlib
import json
import re
from collections import Counter
from dataclasses import dataclass
from enum import Enum
from typing import Any, Callable, Mapping

from training.errors import CorpusPrepError

# ---------------------------------------------------------------------------
# Document model (Section 4) -- minimal, only the fields the directive
# asked for. No "giant metadata framework": one flat frozen dataclass.
# ---------------------------------------------------------------------------


@dataclass(frozen=True)
class RawDocument:
    """One raw input document, before any processing.

    All seven fields are the ones the directive named explicitly; none
    added beyond that (no "genuinely necessary" additional field was
    identified -- everything the pipeline needs to decide ALLOW/REJECT/
    REVIEW, normalize, filter, dedup, and attribute provenance is
    already covered by these seven).
    """

    document_id: str
    text: str
    source_id: str
    source_name: str
    license: str
    language: str
    primary_domain: str


class RejectionReason(Enum):
    """Machine-readable reason a document did not reach ``accepted``.

    One reason per document (the first check that fails wins -- see
    ``prepare_corpus``'s docstring for the exact, fixed check order).
    """

    LICENSE_REJECTED = "license_rejected"
    LICENSE_REVIEW_REQUIRED = "license_review_required"
    EMPTY_AFTER_NORMALIZATION = "empty_after_normalization"
    BELOW_MIN_LENGTH = "below_min_length"
    PATHOLOGICAL_REPETITION = "pathological_repetition"
    EXACT_DUPLICATE = "exact_duplicate"


@dataclass(frozen=True)
class AcceptedDocument:
    """A document that survived every gate, holding enough provenance
    to trace it back to its source (Section 4's "processed
    representation must retain enough provenance" requirement) and
    ``content_hash`` for integrity/reproducibility checks.

    ``text`` here is the NORMALIZED text -- this is what
    ``training.dataset.encode_corpus`` should actually be given (see
    module docstring), not ``RawDocument.text``.
    """

    document_id: str
    text: str
    source_id: str
    source_name: str
    license: str
    language: str
    primary_domain: str
    content_hash: str


@dataclass(frozen=True)
class RejectedDocument:
    """A document that did not reach ``accepted`` -- covers both hard
    rejections and review-required documents (the ``reason`` says
    which; see ``CorpusPrepResult`` for which bucket it lands in).
    Raw (not normalized) text is retained for manual audit/review --
    the entire point of a REVIEW bucket is that a human can look at
    exactly what was excluded and why."""

    document_id: str
    source_id: str
    license: str
    reason: RejectionReason
    text: str


# ---------------------------------------------------------------------------
# License gate (Section 5)
# ---------------------------------------------------------------------------


class LicenseDecision(Enum):
    ALLOW = "allow"
    REJECT = "reject"
    REVIEW = "review"


@dataclass(frozen=True)
class LicensePolicy:
    """An explicit, human-authored mapping from license identifier
    string to :class:`LicenseDecision`. This is a lookup table, not a
    copyright classifier -- exactly what the directive asked for.

    ``default`` governs any license string not present in
    ``decisions``. It defaults to ``REVIEW``, never ``ALLOW`` --
    an unrecognized license is exactly the case the directive says
    "must NOT silently enter the accepted training corpus", so the
    one safe default that satisfies that for every caller (even one
    that forgot to configure a default) is REVIEW, not ALLOW.
    """

    decisions: Mapping[str, LicenseDecision]
    default: LicenseDecision = LicenseDecision.REVIEW

    def decide(self, license_id: str) -> LicenseDecision:
        return self.decisions.get(license_id, self.default)


# ---------------------------------------------------------------------------
# Normalization (Section 6) -- conservative and deterministic.
# ---------------------------------------------------------------------------

_HORIZONTAL_WHITESPACE_RUN = re.compile(r"[ \t]+")


def normalize_text(text: str) -> str:
    """Conservative, deterministic normalization:

    1. CRLF/CR -> LF (newline normalization).
    2. Runs of spaces/tabs collapsed to one space (controlled
       whitespace normalization) -- newlines themselves are NOT
       collapsed, so paragraph structure survives.
    3. Each line stripped of leading/trailing whitespace.
    4. The whole result stripped of leading/trailing whitespace.

    Deliberately NOT done (per the directive's "do not aggressively
    rewrite natural language" / "do not modify code semantics"): no
    case-folding, no Unicode normalization (NFC/NFKC), no punctuation
    changes, no line joining/unwrapping. See this module's "Known
    limitations" note (also in the final report) on why this makes the
    prototype natural-language-oriented, not safe for source code
    corpora as-is (per-line stripping would alter significant
    indentation).
    """
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = _HORIZONTAL_WHITESPACE_RUN.sub(" ", text)
    lines = [line.strip() for line in text.split("\n")]
    return "\n".join(lines).strip()


# ---------------------------------------------------------------------------
# Quality filtering (Section 7) -- simple, deterministic, no ML.
# ---------------------------------------------------------------------------


@dataclass(frozen=True)
class QualityConfig:
    min_length: int = 20
    """Minimum length, in characters, of the NORMALIZED text."""

    min_repeated_tokens_for_pathological: int = 6
    """A document whose normalized text is nothing but the same single
    whitespace-separated token repeated at least this many times is
    rejected as pathological repetition. Deliberately the simplest
    possible trivial check the directive allows ("if trivial to
    detect") -- not a general repetition/compression-ratio detector."""

    def __post_init__(self) -> None:
        if self.min_length < 0:
            raise CorpusPrepError(f"min_length must be >= 0, got {self.min_length}")
        if self.min_repeated_tokens_for_pathological < 1:
            raise CorpusPrepError(
                "min_repeated_tokens_for_pathological must be >= 1, "
                f"got {self.min_repeated_tokens_for_pathological}"
            )


def _is_pathological_repetition(normalized_text: str, config: QualityConfig) -> bool:
    tokens = normalized_text.split()
    return (
        len(tokens) >= config.min_repeated_tokens_for_pathological
        and len(set(tokens)) == 1
    )


def check_quality(
    normalized_text: str, config: QualityConfig
) -> RejectionReason | None:
    """Return the :class:`RejectionReason` this text fails, or
    ``None`` if it passes every quality check.

    An originally-empty and an originally-whitespace-only document are
    indistinguishable once normalized (both normalize to ``""``), so
    both directive examples ("empty document", "whitespace-only
    document") share the single ``EMPTY_AFTER_NORMALIZATION`` reason
    rather than two reasons with no observable difference to assign
    between them.
    """
    if normalized_text == "":
        return RejectionReason.EMPTY_AFTER_NORMALIZATION
    if len(normalized_text) < config.min_length:
        return RejectionReason.BELOW_MIN_LENGTH
    if _is_pathological_repetition(normalized_text, config):
        return RejectionReason.PATHOLOGICAL_REPETITION
    return None


# ---------------------------------------------------------------------------
# Exact deduplication (Section 8)
# ---------------------------------------------------------------------------


def content_hash(normalized_text: str) -> str:
    """SHA-256 hex digest of the normalized text's UTF-8 bytes --
    the directive's suggested stable cryptographic hash; nothing in
    this repository already provides an equivalent (checked: no
    existing content-hashing utility under ``yug/`` or ``training/``
    before adding this)."""
    return hashlib.sha256(normalized_text.encode("utf-8")).hexdigest()


# ---------------------------------------------------------------------------
# Manifest (Section 10)
# ---------------------------------------------------------------------------


@dataclass(frozen=True)
class CorpusManifest:
    schema_version: str
    input_count: int
    accepted_count: int
    rejected_count: int
    review_count: int
    duplicate_count: int
    train_count: int
    validation_count: int
    rejection_reason_counts: dict[str, int]
    source_distribution: dict[str, int]
    license_distribution: dict[str, int]
    language_distribution: dict[str, int]
    domain_distribution: dict[str, int]
    config: dict[str, Any]
    accepted_content_hash: str
    """SHA-256 over the sorted, newline-joined list of every accepted
    document's own ``content_hash`` -- a single reproducibility
    checksum for what actually ended up in the accepted corpus.
    Computed from content hashes (not document_ids), so it is the same
    across two runs even if dedup happened to keep a different
    duplicate's document_id (their normalized text, and therefore
    their content_hash, is identical by construction)."""

    def to_dict(self) -> dict[str, Any]:
        return {
            "schema_version": self.schema_version,
            "input_count": self.input_count,
            "accepted_count": self.accepted_count,
            "rejected_count": self.rejected_count,
            "review_count": self.review_count,
            "duplicate_count": self.duplicate_count,
            "train_count": self.train_count,
            "validation_count": self.validation_count,
            "rejection_reason_counts": self.rejection_reason_counts,
            "source_distribution": self.source_distribution,
            "license_distribution": self.license_distribution,
            "language_distribution": self.language_distribution,
            "domain_distribution": self.domain_distribution,
            "config": self.config,
            "accepted_content_hash": self.accepted_content_hash,
        }

    def to_json(self) -> str:
        """Deterministic JSON: sorted keys, no timestamps, no other
        nondeterministic values -- two runs on the same input with the
        same configuration produce byte-identical output."""
        return json.dumps(self.to_dict(), sort_keys=True, indent=2)


def _sorted_counts(items: list[str]) -> dict[str, int]:
    counts = Counter(items)
    return {key: counts[key] for key in sorted(counts)}


# ---------------------------------------------------------------------------
# The pipeline itself
# ---------------------------------------------------------------------------


@dataclass(frozen=True)
class CorpusPrepResult:
    accepted: list[AcceptedDocument]
    rejected: list[RejectedDocument]
    review: list[RejectedDocument]
    train: list[AcceptedDocument]
    validation: list[AcceptedDocument]
    manifest: CorpusManifest


def prepare_corpus(
    documents: list[RawDocument],
    *,
    license_policy: LicensePolicy,
    quality_config: QualityConfig | None = None,
    val_fraction: float,
    seed: int,
    text_normalizer: Callable[[RawDocument], str] | None = None,
) -> CorpusPrepResult:
    """Run the full Corpus Preparation Prototype pipeline.

    Fixed check order per document (first failing check wins -- this
    is deliberate and documented, not an implementation accident):

        1. License gate (on ``RawDocument.license``, raw text untouched)
        2. Normalization (on ``RawDocument.text``)
        3. Quality filter (on the normalized text)
        4. Exact deduplication (across all documents that passed 1-3,
           in ``documents`` input order -- first occurrence of a given
           normalized text wins as accepted, every later occurrence is
           rejected as EXACT_DUPLICATE)

    Then: the surviving accepted documents' (normalized) text is
    passed to ``training.split.split_sentences`` (reused, not
    reimplemented) for the deterministic train/validation split.

    ``text_normalizer`` (optional, backward-compatible): a function
    ``RawDocument -> str`` used INSTEAD of :func:`normalize_text` at
    step 2. ``None`` (the default) keeps the original prose-oriented
    behaviour byte-for-byte. It exists because ``normalize_text``
    strips per-line indentation and collapses space runs, which is
    destructive for source code and markdown; content-type-aware
    callers (see ``training.experiments.representative_mini_corpus``)
    supply a whitespace-preserving normalizer for those documents.

    Raises:
        CorpusPrepError: ``documents`` is empty.
    """
    if not documents:
        raise CorpusPrepError("documents must contain at least one entry")
    quality_config = quality_config or QualityConfig()

    accepted: list[AcceptedDocument] = []
    rejected: list[RejectedDocument] = []
    review: list[RejectedDocument] = []
    seen_hashes: set[str] = set()
    duplicate_count = 0

    for doc in documents:
        decision = license_policy.decide(doc.license)

        if decision is LicenseDecision.REJECT:
            rejected.append(
                RejectedDocument(
                    document_id=doc.document_id,
                    source_id=doc.source_id,
                    license=doc.license,
                    reason=RejectionReason.LICENSE_REJECTED,
                    text=doc.text,
                )
            )
            continue

        if decision is LicenseDecision.REVIEW:
            review.append(
                RejectedDocument(
                    document_id=doc.document_id,
                    source_id=doc.source_id,
                    license=doc.license,
                    reason=RejectionReason.LICENSE_REVIEW_REQUIRED,
                    text=doc.text,
                )
            )
            continue

        normalized = normalize_text(doc.text) if text_normalizer is None else text_normalizer(doc)
        quality_failure = check_quality(normalized, quality_config)
        if quality_failure is not None:
            rejected.append(
                RejectedDocument(
                    document_id=doc.document_id,
                    source_id=doc.source_id,
                    license=doc.license,
                    reason=quality_failure,
                    text=doc.text,
                )
            )
            continue

        doc_hash = content_hash(normalized)
        if doc_hash in seen_hashes:
            duplicate_count += 1
            rejected.append(
                RejectedDocument(
                    document_id=doc.document_id,
                    source_id=doc.source_id,
                    license=doc.license,
                    reason=RejectionReason.EXACT_DUPLICATE,
                    text=doc.text,
                )
            )
            continue
        seen_hashes.add(doc_hash)

        accepted.append(
            AcceptedDocument(
                document_id=doc.document_id,
                text=normalized,
                source_id=doc.source_id,
                source_name=doc.source_name,
                license=doc.license,
                language=doc.language,
                primary_domain=doc.primary_domain,
                content_hash=doc_hash,
            )
        )

    train, validation = _split_accepted(accepted, val_fraction=val_fraction, seed=seed)

    manifest = _build_manifest(
        documents=documents,
        accepted=accepted,
        rejected=rejected,
        review=review,
        duplicate_count=duplicate_count,
        train=train,
        validation=validation,
        quality_config=quality_config,
        val_fraction=val_fraction,
        seed=seed,
    )

    return CorpusPrepResult(
        accepted=accepted,
        rejected=rejected,
        review=review,
        train=train,
        validation=validation,
        manifest=manifest,
    )


def _split_accepted(
    accepted: list[AcceptedDocument], *, val_fraction: float, seed: int
) -> tuple[list[AcceptedDocument], list[AcceptedDocument]]:
    """Deterministic split, reusing ``training.split.split_sentences``
    rather than reimplementing splitting logic. That function operates
    on ``list[str]``; texts are guaranteed unique here (deduplication
    already ran), so mapping each returned string back to its
    originating :class:`AcceptedDocument` by exact text match is safe
    and unambiguous."""
    by_text = {doc.text: doc for doc in accepted}
    assert len(by_text) == len(accepted), (
        "accepted documents must have unique text at this point -- "
        "deduplication should have already guaranteed this"
    )

    if len(accepted) < 2:
        # split_sentences requires n_val >= 1 and at least 1 remaining
        # training sentence, i.e. at least 2 inputs. A one-(or-zero)-
        # document accepted corpus cannot be meaningfully split; every
        # accepted document goes to train, validation is empty. This
        # is a fixture/small-corpus edge case, not a silent departure
        # from determinism -- the same input always produces this same
        # outcome.
        return list(accepted), []

    train_texts, val_texts = split_sentences(
        [doc.text for doc in accepted], val_fraction=val_fraction, seed=seed
    )
    return [by_text[t] for t in train_texts], [by_text[t] for t in val_texts]


def _build_manifest(
    *,
    documents: list[RawDocument],
    accepted: list[AcceptedDocument],
    rejected: list[RejectedDocument],
    review: list[RejectedDocument],
    duplicate_count: int,
    train: list[AcceptedDocument],
    validation: list[AcceptedDocument],
    quality_config: QualityConfig,
    val_fraction: float,
    seed: int,
) -> CorpusManifest:
    rejection_reason_counts = _sorted_counts(
        [r.reason.value for r in rejected] + [r.reason.value for r in review]
    )
    accepted_hashes_sorted = sorted(doc.content_hash for doc in accepted)
    accepted_content_hash = hashlib.sha256(
        "\n".join(accepted_hashes_sorted).encode("utf-8")
    ).hexdigest()

    return CorpusManifest(
        schema_version=_SCHEMA_VERSION,
        input_count=len(documents),
        accepted_count=len(accepted),
        rejected_count=len(rejected),
        review_count=len(review),
        duplicate_count=duplicate_count,
        train_count=len(train),
        validation_count=len(validation),
        rejection_reason_counts=rejection_reason_counts,
        source_distribution=_sorted_counts([d.source_id for d in documents]),
        license_distribution=_sorted_counts([d.license for d in documents]),
        language_distribution=_sorted_counts([d.language for d in documents]),
        domain_distribution=_sorted_counts([d.primary_domain for d in documents]),
        config={
            "val_fraction": val_fraction,
            "seed": seed,
            "min_length": quality_config.min_length,
            "min_repeated_tokens_for_pathological": (
                quality_config.min_repeated_tokens_for_pathological
            ),
        },
        accepted_content_hash=accepted_content_hash,
    )

===== FILE: training\split.py =====
"""NB-08: deterministic train/validation split at the sentence level.

Deliberately its own module, not part of ``training/dataset.py``: this
function has no PyTorch dependency (only Python's ``random``), and
keeping it separate from ``dataset.py`` (which imports ``torch`` at
module level) means it can actually be imported and tested in any
environment, mirroring the same reasoning already applied to
``training/config.py`` and ``training/errors.py``.

Splitting happens at the SENTENCE level, before tokenization, not at
the flat-token-stream level -- two reasons:

1. A tokenizer trained on the full corpus (train + validation text
   together) would let its learned merge vocabulary "see" validation
   text before the model ever gets evaluated on it -- a real, if
   subtle, form of leakage. Splitting sentences first, and training
   the tokenizer ONLY on the training sentences (see
   ``training/experiments/nb08_experiment.py``), avoids this: any
   byte-pair patterns unique to validation sentences are genuinely
   unseen by the tokenizer, making validation loss a more honest
   generalization signal.
2. Splitting an already-tokenized flat stream at an arbitrary token
   index risks cutting a sentence in half across the train/val
   boundary, silently leaking a is-this-sentence-continuing signal
   into whichever split got the other half. Splitting whole sentences
   avoids this entirely.
"""

from __future__ import annotations

import random

from training.errors import DatasetError


def split_sentences(
    sentences: list[str], val_fraction: float, seed: int
) -> tuple[list[str], list[str]]:
    """Deterministically shuffle ``sentences`` (using a
    ``random.Random(seed)`` instance local to this call -- this does
    NOT touch the global ``random`` module state, so it cannot
    interfere with :func:`training.determinism.seed_everything`'s
    seeding of anything else) and split off ``val_fraction`` of them
    (rounded down, at least 1) as the validation set. The remainder is
    the training set, in shuffled order.

    Raises:
        DatasetError: ``sentences`` is empty, ``val_fraction`` is not
            in ``(0, 1)``, or the split would leave zero training
            sentences.
    """
    if not sentences:
        raise DatasetError("sentences must contain at least one entry")
    if not (0 < val_fraction < 1):
        raise DatasetError(f"val_fraction must be in (0, 1), got {val_fraction}")

    shuffled = sentences.copy()
    random.Random(seed).shuffle(shuffled)

    n_val = max(1, int(len(shuffled) * val_fraction))
    if n_val >= len(shuffled):
        raise DatasetError(
            f"val_fraction={val_fraction} on {len(shuffled)} sentences would "
            f"leave zero training sentences"
        )

    val_sentences = shuffled[:n_val]
    train_sentences = shuffled[n_val:]
    return train_sentences, val_sentences


===== FILE: training\experiments\real_corpus_experiment.py =====
"""First Real-Corpus Experiment.

NOT a milestone: no NB-13 is assigned, no roadmap file is touched by
this script or its execution. Chief-architect-authorized controlled
experiment, same status as
``training.experiments.scaling_scale3_budget_extension``.

OBJECTIVE: test whether real English prose can flow through the
already-verified pipeline (corpus_prep -> tokenizer -> dataset ->
Trainer -> evaluate -> checkpoint -> generation) unchanged, and
measure how the existing Scale-3 architecture behaves on it.

============================================================
DATASET ACQUISITION: bounded, pinned-revision streaming read
============================================================

Source: ``common-pile/wikimedia_filtered`` (Hugging Face), the
"filtered" Wikimedia subset of Common Pile v0.1. Official Wikimedia
wikis are CC BY-SA per the dataset card; the per-record license string
in the data itself is what this module actually maps (see below), not
the card's summary.

Mechanism (:func:`acquire_common_pile_wikimedia_sample`):
  - Python standard library only (``urllib.request``, ``gzip``,
    ``json``) -- NO new dependency, no ``datasets``/``pyarrow``.
  - Streams ``wikimedia-0000.json.gz`` (one of 35 shards, each
    roughly 500-800 MB compressed) from the dataset's files at a
    PINNED git revision, decompressing on the fly, one JSON record
    per line. It stops reading, and closes the connection, the moment
    ``max_documents`` qualifying records have been collected -- the
    remainder of the shard is never requested by this code.
  - Bounded a second way: ``max_records_scanned`` caps how many lines
    may be examined, so a shard that yields few qualifying records
    cannot turn into an unbounded read.
  - Deterministic: the result is a pure function of (pinned revision,
    ``shard_indices``, ``main_namespace_only``, ``max_documents``) --
    records are taken in shard/line order, with no randomness.
  - Selection rule: only main-namespace pages (``metadata.namespace``
    == "0", i.e. encyclopedic/article pages) by default, so user pages,
    talk pages and project-administration pages are not counted as
    "real prose/reference text". This is a schema-based rule, not a
    quality judgement; ``corpus_prep`` still applies its own quality
    filter and exact deduplication afterwards.

Record schema actually used (read from the official dataset card /
viewer, not assumed): top-level ``id``, ``text``, ``source``;
``metadata.license``, ``metadata.namespace``, and (optional, kept as
provenance when present) ``metadata.title``, ``metadata.url``,
``metadata.provenance``, ``metadata.wiki``, ``metadata.authors``,
plus top-level ``created``.

License handling (the existing gate is NOT bypassed):
  - The real per-record value is
    ``"Creative Commons - Attribution Share-Alike -
    https://creativecommons.org/licenses/by-sa/4.0/"``. It is mapped
    by EXACT string match to YUG's ``CC-BY-SA-4.0`` identifier.
  - Any other license string is passed through to ``corpus_prep``
    VERBATIM, unmapped, so the existing policy quarantines it to
    REVIEW instead of it being relabelled as allowed.
  - A record with no license string is skipped and counted; a license
    is never invented for it.
  - The dataset card itself warns some licenses may be assigned
    incorrectly upstream ("license laundering"); this module cannot
    detect that, and does not claim to.

Provenance: ``RawDocument`` has no slot for title/URL/locator, and
``corpus_prep`` is deliberately not being redesigned here. The full
per-document provenance therefore travels in
:class:`AcquisitionReport.provenance`, keyed by ``document_id``
(``"<source>:<record id>"``), and is written into the result artifact
under ``"acquisition"``. Nothing downloaded is stored in the repository.

VERIFICATION STATUS OF ACQUISITION: written against the schema read
from the official dataset card, and exercised in tests with an
injected in-memory gzip stream (so the real decompression, JSON-lines
parsing, mapping, stopping and error paths run) -- but the LIVE
network read has NOT been executed by the author of this change,
because the authoring sandbox has no outbound network. First real
execution is Rai's.

Everything downstream of acquisition --
:func:`run_real_corpus_experiment` and its helpers -- takes a plain
``list[RawDocument]`` and does not know where the documents came from.

VERIFICATION STATUS OF THE REST: this module has no PyTorch dependency
at import time (only inside ``run_real_corpus_experiment``, matching
``scaling_scale3_budget_extension.py``'s convention). The torch-free
parts (acquisition, ``compute_tokenizer_stats``,

PyTorch. The training/evaluation/generation stages require PyTorch and
were verified independently on Rai's Windows environment against a tiny

FROZEN CONTROLS reused via import, never duplicated or modified (same

  - architecture: hidden_size=48, num_layers=2, num_heads=4, ffn_size=96
  - sequence_length=16, context_length=32

    batch_size=4 (``training.experiments.nb08_experiment.build_training_config``'s

  - tokenizer vocab_size target=384
    (``training.experiments.nb08_experiment.TOKENIZER_VOCAB_SIZE``,

  - baselines: ``training.experiments.nb08_baseline.{uniform_baseline_loss,
    unigram_baseline_loss}``
  - decode-safety: ``training.experiments.nb08_experiment.safe_decode_generation``
  - sustained-overfitting stop condition: copied verbatim from

@dataclass(frozen=True)
class AcquisitionReport:
    """What acquisition actually did -- recorded, not assumed. Written
    into the result artifact under ``"acquisition"``.

    ``provenance`` maps ``document_id`` to the real source metadata
    (title, URL, locator, wiki, namespace, authors, the source's own
    license string, ...) that ``RawDocument`` has no fields for. Only
    metadata actually present in the source record is included."""

    dataset_id: str
    dataset_revision: str
    shard_files: tuple[str, ...]
    max_documents: int
    main_namespace_only: bool
    records_scanned: int
    records_selected: int
    skipped: dict[str, int]
    compressed_bytes_read: int
    provenance: dict[str, dict[str, Any]]

    def to_dict(self) -> dict[str, Any]:
        return {
            "dataset_id": self.dataset_id,
            "dataset_revision": self.dataset_revision,
            "shard_files": list(self.shard_files),
            "max_documents": self.max_documents,
            "main_namespace_only": self.main_namespace_only,
            "records_scanned": self.records_scanned,
            "records_selected": self.records_selected,
            "skipped": dict(sorted(self.skipped.items())),
            "compressed_bytes_read": self.compressed_bytes_read,
            "provenance": self.provenance,
        }


class _CountingReader:
    """Minimal read-only wrapper that counts compressed bytes actually
    pulled off the wire -- evidence that the read really was bounded."""

    def __init__(self, raw: BinaryIO) -> None:
        self._raw = raw
        self.bytes_read = 0

    def read(self, size: int = -1) -> bytes:
        data = self._raw.read(size)
        self.bytes_read += len(data)
        return data


def _open_shard_stream(url: str) -> BinaryIO:
    """Default network opener: a plain HTTPS GET, streamed. Only this
    function touches the network, so tests replace it wholesale."""
    request = urllib.request.Request(url, headers={"User-Agent": _USER_AGENT})
    return urllib.request.urlopen(request, timeout=_HTTP_TIMEOUT_SECONDS)  # noqa: S310


_SKIP_MALFORMED = "malformed"
_SKIP_MISSING_LICENSE = "missing_license"
_SKIP_WRONG_NAMESPACE = "wrong_namespace"
_SKIP_DUPLICATE_ID = "duplicate_id"


def _present_str(value: object) -> str | None:
    return value if isinstance(value, str) and value.strip() else None


def _map_record(
    record: object, *, shard_file: str, main_namespace_only: bool
) -> tuple[str, tuple[RawDocument, dict[str, Any]] | None]:
    """Map one source record to ``(status, payload)``. ``status`` is
    ``"ok"`` (payload = (RawDocument, provenance)) or a skip reason
    (payload None). Never invents a value for a missing field."""
    if not isinstance(record, dict):
        return _SKIP_MALFORMED, None

    record_id = _present_str(record.get("id"))
    source = _present_str(record.get("source"))
    text = record.get("text")
    metadata = record.get("metadata")
    if record_id is None or source is None or not isinstance(text, str) or not isinstance(metadata, dict):
        return _SKIP_MALFORMED, None

    namespace = metadata.get("namespace")
    if main_namespace_only:
        if namespace is None:
            return _SKIP_MALFORMED, None
        if str(namespace) != "0":
            return _SKIP_WRONG_NAMESPACE, None

    license_source_string = _present_str(metadata.get("license"))
    if license_source_string is None:
        return _SKIP_MISSING_LICENSE, None
    # Exact-match mapping only; an unrecognized string is kept verbatim
    # so the existing license gate quarantines it (never relabelled).
    license_id = _SOURCE_LICENSE_TO_YUG_LICENSE.get(license_source_string, license_source_string)

    document_id = f"{source}:{record_id}"
    document = RawDocument(
        document_id=document_id,
        text=text,
        source_id=source,
        source_name=f"{COMMON_PILE_DATASET_ID}@{COMMON_PILE_DATASET_REVISION[:12]}",
        license=license_id,
        language=DATASET_LANGUAGE,
        primary_domain=source.rsplit("/", 1)[-1],
    )

    provenance: dict[str, Any] = {
        "dataset": COMMON_PILE_DATASET_ID,
        "revision": COMMON_PILE_DATASET_REVISION,
        "shard_file": shard_file,
        "record_id": record_id,
        "source": source,
        "license_source_string": license_source_string,
        "license_yug_id": license_id,
    }
    for key in ("title", "url", "provenance", "wiki"):
        value = _present_str(metadata.get(key))
        if value is not None:
            provenance["provenance_locator" if key == "provenance" else key] = value
    if namespace is not None:
        provenance["namespace"] = str(namespace)
    if isinstance(metadata.get("authors"), list):
        provenance["authors"] = metadata["authors"]
    created = _present_str(record.get("created"))
    if created is not None:
        provenance["created"] = created
    return "ok", (document, provenance)


def acquire_common_pile_wikimedia_sample(
    max_documents: int = 1000,
    *,
    shard_indices: tuple[int, ...] = (0,),
    main_namespace_only: bool = True,
    max_records_scanned: int = 100_000,
    max_unusable_records: int = 100,
    open_stream: Callable[[str], BinaryIO] | None = None,
) -> tuple[list[RawDocument], AcquisitionReport]:
    """Acquire a small, deterministic, bounded sample of
    ``common-pile/wikimedia_filtered`` as :class:`RawDocument` objects.

    See the module docstring for the mechanism, selection rule, schema,
    and license handling. ``open_stream`` exists so tests can supply an
    in-memory stream; production callers leave it ``None``.

    Raises:
        RealCorpusAcquisitionError: bad arguments; any network/HTTP/
            decompression/decoding failure; a line that is not valid
            JSON; too many unusable records (schema mismatch or missing
            licenses -- refusing to bypass the license gate); the scan
            cap reached, or every requested shard exhausted, before
            ``max_documents`` qualifying records were found.
    """
    if max_documents < 1:
        raise RealCorpusAcquisitionError(f"max_documents must be >= 1, got {max_documents}")
    if not shard_indices:
        raise RealCorpusAcquisitionError("shard_indices must contain at least one shard index")
    for index in shard_indices:
        if not 0 <= index < _SHARD_COUNT:
            raise RealCorpusAcquisitionError(
                f"shard index {index} out of range (valid: 0..{_SHARD_COUNT - 1})"
            )
    if max_records_scanned < 1 or max_unusable_records < 0:
        raise RealCorpusAcquisitionError("max_records_scanned must be >= 1 and max_unusable_records >= 0")

    opener = open_stream if open_stream is not None else _open_shard_stream
    documents: list[RawDocument] = []
    provenance: dict[str, dict[str, Any]] = {}
    seen_ids: set[str] = set()
    skipped: Counter[str] = Counter()
    shards_opened: list[str] = []
    records_scanned = 0
    compressed_bytes_read = 0

    try:
        for index in shard_indices:
            if len(documents) >= max_documents:
                break
            shard_file = f"wikimedia-{index:04d}.json.gz"
            url = _SHARD_URL_TEMPLATE.format(
                dataset=COMMON_PILE_DATASET_ID, revision=COMMON_PILE_DATASET_REVISION, index=index
            )
            shards_opened.append(shard_file)

            with contextlib.closing(opener(url)) as raw:
                counting = _CountingReader(raw)
                try:
                    with io.TextIOWrapper(
                        gzip.GzipFile(fileobj=counting, mode="rb"), encoding="utf-8"
                    ) as lines:
                        for line in lines:
                            if not line.strip():
                                continue
                            if records_scanned >= max_records_scanned:
                                raise RealCorpusAcquisitionError(
                                    f"scan cap reached: examined {records_scanned} records without "
                                    f"collecting {max_documents} qualifying documents "
                                    f"(collected {len(documents)}); raise max_records_scanned, "
                                    f"add shard_indices, or lower max_documents"
                                )
                            records_scanned += 1
                            try:
                                record = json.loads(line)
                            except json.JSONDecodeError as exc:
                                raise RealCorpusAcquisitionError(
                                    f"{shard_file}: record #{records_scanned} is not valid JSON "
                                    f"({exc}); the source format differs from the expected "
                                    f"one-JSON-object-per-line layout"
                                ) from exc

                            status, payload = _map_record(
                                record, shard_file=shard_file, main_namespace_only=main_namespace_only
                            )
                            if status != "ok":
                                skipped[status] += 1
                                if skipped[_SKIP_MALFORMED] + skipped[_SKIP_MISSING_LICENSE] > max_unusable_records:
                                    raise RealCorpusAcquisitionError(
                                        f"{shard_file}: more than {max_unusable_records} records were "
                                        f"unusable (malformed={skipped[_SKIP_MALFORMED]}, "
                                        f"missing_license={skipped[_SKIP_MISSING_LICENSE]}) -- either "
                                        f"the dataset schema differs from what this code expects or "
                                        f"license metadata is absent; refusing to bypass the license gate"
                                    )
                                continue

                            document, record_provenance = payload
                            if document.document_id in seen_ids:
                                skipped[_SKIP_DUPLICATE_ID] += 1
                                continue
                            seen_ids.add(document.document_id)
                            documents.append(document)
                            provenance[document.document_id] = record_provenance
                            if len(documents) >= max_documents:
                                break
                finally:
                    compressed_bytes_read += counting.bytes_read
    except RealCorpusAcquisitionError:
        raise
    except (OSError, EOFError, http.client.HTTPException, zlib.error, UnicodeDecodeError) as exc:
        status_code = getattr(exc, "code", None)
        detail = f"HTTP {status_code}: " if status_code is not None else ""
        raise RealCorpusAcquisitionError(
            f"could not read {COMMON_PILE_DATASET_ID}@{COMMON_PILE_DATASET_REVISION[:12]} "
            f"({shards_opened[-1] if shards_opened else 'no shard opened'}): "
            f"{detail}{type(exc).__name__}: {exc}"
        ) from exc

    if len(documents) < max_documents:
        raise RealCorpusAcquisitionError(
            f"requested shards exhausted after {records_scanned} records with only "
            f"{len(documents)} of {max_documents} qualifying documents "
            f"(skipped: {dict(sorted(skipped.items()))}); add shard_indices or lower max_documents"
        )

    report = AcquisitionReport(
        dataset_id=COMMON_PILE_DATASET_ID,
        dataset_revision=COMMON_PILE_DATASET_REVISION,
        shard_files=tuple(shards_opened),
        max_documents=max_documents,
        main_namespace_only=main_namespace_only,
        records_scanned=records_scanned,
        records_selected=len(documents),
        skipped=dict(skipped),
        compressed_bytes_read=compressed_bytes_read,
        provenance=provenance,
    )
    return documents, report


def load_common_pile_wikimedia_sample(max_documents: int = 1000) -> list[RawDocument]:
    """Convenience wrapper returning only the documents. Use
    :func:`acquire_common_pile_wikimedia_sample` when the
    :class:`AcquisitionReport` (provenance, counts, bytes read) is
    needed -- ``__main__`` does, so it lands in the result artifact."""
    documents, _report = acquire_common_pile_wikimedia_sample(max_documents)
    return documents


def _count_words(text: str) -> int:
    """The one documented word-count definition this experiment uses:
    whitespace-split token count on already-normalized text. Simple
    and deterministic by construction (``str.split()`` with no
    arguments splits on any run of whitespace and discards empty
    strings) -- not a linguistic tokenizer, deliberately."""
    return len(text.split())


def compute_tokenizer_stats(tokenizer, texts: list[str]) -> dict:
    """Tokenizer efficiency/health statistics over ``texts`` (already-
    normalized, accepted document text). Genuinely new: nothing
    existing in this repository already computes this.

    Does NOT report an unknown-token count/rate: verified directly
    (``yug/tokenizer/bpe.py``'s own docstring) that
    ``ByteLevelBpeTokenizer`` has no ``<unk>`` concept at all -- every
    byte is already a valid base symbol, so an unknown-token fallback
    would be dead code, per that module's own documented design
    decision. Inventing a metric here for a concept the tokenizer
    doesn't have would be exactly what this directive's TOKENIZER
    section forbids, so the key is omitted rather than reported as an
    always-zero placeholder.
    """
    from yug.tokenizer.errors import DecodeError

    total_characters = 0
    total_words = 0
    total_tokens = 0
    encoding_failures = 0
    decoding_failures = 0

    for text in texts:
        total_characters += len(text)
        total_words += _count_words(text)
        try:
            token_ids = tokenizer.encode(text)
        except Exception:  # noqa: BLE001 - record, do not crash the stat pass
            encoding_failures += 1
            continue
        total_tokens += len(token_ids)
        try:
            tokenizer.decode(token_ids)
        except DecodeError:
            # Not expected for a direct encode->decode round trip (see
            # tokenizer's own DecodeError docstring: this is for
            # arbitrary/generated token sequences that don't land on a
            # valid UTF-8 byte boundary, not for decoding a sequence
            # the tokenizer itself just produced from valid text) --
            # measured honestly rather than assumed to always be 0.
            decoding_failures += 1

    return {
        "tokenizer_identity": tokenizer.identity,
        "tokenizer_vocab_size": tokenizer.vocab_size,
        "document_count": len(texts),
        "total_characters": total_characters,
        "total_words": total_words,
        "total_tokens": total_tokens,
        "tokens_per_word": (total_tokens / total_words) if total_words else None,
        "tokens_per_character": (total_tokens / total_characters) if total_characters else None,
        "encoding_failures": encoding_failures,
        "decoding_failures": decoding_failures,
        "unknown_token_count": None,
        "unknown_token_rate": None,
        "unknown_token_note": (
            "not applicable: ByteLevelBpeTokenizer has no <unk> concept "
            "(every byte is a valid base symbol) -- see yug/tokenizer/bpe.py"
        ),
    }


def _sequence_utilization(num_sequences: int, sequence_length: int, total_tokens: int) -> float | None:
    """Fraction of encoded tokens that actually ended up inside a
    training sequence. ``make_sequences`` truncates a final partial
    chunk (see ``training/dataset.py``), so this is always <= 1.0 and
    directly derivable from its own documented chunking behavior --
    not a new metric definition, just this experiment's own bookkeeping
    of an existing function's already-documented truncation."""
    if total_tokens == 0:
        return None
    return (num_sequences * (sequence_length + 1)) / total_tokens


def train_tokenizer_on_train_partition(prep, *, vocab_size: int = TOKENIZER_VOCAB_SIZE):
    """Train the EXISTING ``ByteLevelBpeTokenizer`` on the TRAINING
    partition of a ``CorpusPrepResult`` only. Validation documents are
    never passed to tokenizer training (Directive: "Validation
    documents MUST NOT participate in tokenizer training").

    Behaviour-preserving extraction of what
    :func:`run_real_corpus_experiment` already did inline; pulled out
    only so this property is testable without PyTorch. ``vocab_size``
    is a target, not a guarantee -- BPE can stop short if the corpus
    has no more mergeable pairs -- so callers must report the actual
    ``tokenizer.vocab_size`` (``run_real_corpus_experiment`` does).
    """
    from yug.tokenizer.bpe import ByteLevelBpeTokenizer

    train_texts = [doc.text for doc in prep.train]
    return ByteLevelBpeTokenizer.train(
        corpus=train_texts, vocab_size=vocab_size, special_tokens=("bos", "eos")
    )

def _check_state_dict_finite(state_dict) -> tuple[bool, str]:
    """Copied verbatim from
    ``training.experiments.scaling_scale3_budget_extension._check_state_dict_finite``
    (itself copied from ``training.experiments.nb12_checkpoint_closure``'s
    equivalent helper) -- not re-derived, not imported across
    independent experiment scripts, matching both files' own stated
    reason for keeping this local."""
    import torch

    bad = [name for name, t in state_dict.items() if not torch.isfinite(t).all()]

def run_real_corpus_experiment(documents: list[RawDocument]) -> dict:
    """Run the complete first-real-corpus pipeline on ``documents``.

    Reuses ``training.corpus_prep.prepare_corpus`` for the ENTIRE
    provenance/license/normalize/filter/dedup/split stage (Directive:
    "Do not bypass training/corpus_prep.py") -- the document-level
    train/validation split happens there, before any tokenization, so
    "content from the same source document must never appear in both
    train and validation" is corpus_prep's own already-tested
    guarantee, not reimplemented here.

    Raises:
        CorpusPrepError: (from ``prepare_corpus``) ``documents`` is
            empty, or too few documents survive to produce both a
            non-empty train and validation split.
    """
    import torch

    from training.checkpoint import save_checkpoint
    from training.dataset import encode_corpus, iter_batches, make_sequences
    from training.determinism import seed_everything
    from training.evaluate import evaluate_checkpoint
    from training.experiments.nb08_baseline import unigram_baseline_loss, uniform_baseline_loss
    from training.experiments.nb08_experiment import safe_decode_generation
    from training.loop import Trainer
    from yug.native.config import ModelConfig
    from yug.native.generation import generate
    from yug.native.model import YugTransformer

    result: dict = {
        "experiment": "first_real_corpus_experiment",
        "not_a_milestone": True,
        "nb13_assigned": False,
        "dataset_source": "Common Pile - Wikimedia Filtered",
        "dataset_license": WIKIMEDIA_LICENSE,
        "requested_document_count": len(documents),
        "stop_reason": None,
    }

    prep = prepare_corpus(
        documents,
        license_policy=REAL_CORPUS_LICENSE_POLICY,
        val_fraction=VAL_FRACTION,
        seed=SEED,
    )
    result["corpus_manifest"] = prep.manifest.to_dict()

    train_texts = [doc.text for doc in prep.train]
    val_texts = [doc.text for doc in prep.validation]

    tokenizer = train_tokenizer_on_train_partition(prep)
    result["tokenizer_training"] = {
        "trained_on": "train_partition_only",
        "train_document_count": len(prep.train),
        "requested_vocab_size": TOKENIZER_VOCAB_SIZE,
        "actual_vocab_size": tokenizer.vocab_size,
        "tokenizer_identity": tokenizer.identity,
    }
    result["tokenizer_stats"] = {
        "train": compute_tokenizer_stats(tokenizer, train_texts),
        "validation": compute_tokenizer_stats(tokenizer, val_texts),
    }

    train_token_ids = encode_corpus(tokenizer, train_texts)
    val_token_ids = encode_corpus(tokenizer, val_texts)
    train_sequences = make_sequences(train_token_ids, sequence_length=SEQUENCE_LENGTH)
    val_sequences = make_sequences(val_token_ids, sequence_length=SEQUENCE_LENGTH)

    result["sequence_utilization"] = {
        "train": _sequence_utilization(len(train_sequences), SEQUENCE_LENGTH, len(train_token_ids)),
        "validation": _sequence_utilization(len(val_sequences), SEQUENCE_LENGTH, len(val_token_ids)),
    }

    if not train_sequences or not val_sequences:
        result["stop_reason"] = (
            f"insufficient tokens for at least one sequence: "
            f"train_sequences={len(train_sequences)}, val_sequences={len(val_sequences)}"
        )
        return result

    model_config = ModelConfig.from_tokenizer(
        tokenizer,
        hidden_size=HIDDEN_SIZE,
        num_layers=NUM_LAYERS,
        num_heads=NUM_HEADS,
        ffn_size=FFN_SIZE,
        context_length=CONTEXT_LENGTH,
    )
    training_config = build_training_config()

    seed_everything(SEED)
    model = YugTransformer(model_config)
    parameter_count = model.count_parameters()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=training_config.learning_rate, weight_decay=training_config.weight_decay
    )
    trainer = Trainer(model, optimizer, training_config, model_config, tokenizer.identity)

    train_batches = iter_batches(train_sequences, training_config.batch_size)
    val_batches = iter_batches(val_sequences, training_config.batch_size)

    def _current_losses() -> tuple[float, float]:
        return trainer.evaluate(train_batches), trainer.evaluate(val_batches)

    import tempfile
    from pathlib import Path

    checkpoint_root = Path(tempfile.mkdtemp(prefix="real_corpus_experiment_"))

    training_curve: list[dict] = []
    checkpoints_saved: dict[int, dict] = {}
    generation_samples: dict[int, dict[str, str]] = {}
    stop_reason: str | None = None

    initial_train_loss, initial_val_loss = _current_losses()
    training_curve.append({"step": 0, "train_loss": initial_train_loss, "val_loss": initial_val_loss})
    if not (math.isfinite(initial_train_loss) and math.isfinite(initial_val_loss)):
        stop_reason = "non-finite loss at step 0 (before any training)"

    batch_idx = 0
    while stop_reason is None and trainer.step < training_config.max_steps:
        input_ids, target_ids = train_batches[batch_idx % len(train_batches)]
        step_train_loss = trainer.train_step(input_ids, target_ids)
        batch_idx += 1

        if not math.isfinite(step_train_loss):
            stop_reason = f"non-finite training-step loss at step {trainer.step}: {step_train_loss}"
            break

        if trainer.step not in EVAL_CHECKPOINTS:
            continue

        train_loss, val_loss = _current_losses()
        training_curve.append({"step": trainer.step, "train_loss": train_loss, "val_loss": val_loss})

        if not (math.isfinite(train_loss) and math.isfinite(val_loss)):
            stop_reason = f"non-finite loss at step {trainer.step}: train={train_loss} val={val_loss}"
            break

        if trainer.step in RETAINED_CHECKPOINTS:
            ckpt_dir = checkpoint_root / f"step_{trainer.step}"
            save_checkpoint(
                ckpt_dir,
                model=model,
                optimizer=optimizer,
                model_config=model_config,
                training_config=training_config,
                step=trainer.step,
                tokenizer_identity=tokenizer.identity,
            )
            raw_state = torch.load(ckpt_dir / "state.pt", map_location="cpu")
            state_ok, state_detail = _check_state_dict_finite(raw_state["model"])

            eval_ok, eval_detail, eval_loss, eval_ppl = True, None, None, None
            try:
                eval_report = evaluate_checkpoint(
                    str(ckpt_dir),
                    tokenizer,
                    val_texts,
                    sequence_length=SEQUENCE_LENGTH,
                    batch_size=training_config.batch_size,
                    seed=SEED,
                )
                eval_loss, eval_ppl = eval_report.loss, eval_report.perplexity
                if not (math.isfinite(eval_loss) and math.isfinite(eval_ppl)):
                    eval_ok = False
                    eval_detail = f"non-finite: loss={eval_loss} perplexity={eval_ppl}"
            except Exception as exc:  # noqa: BLE001 - record, don't crash the experiment
                eval_ok = False
                eval_detail = f"{type(exc).__name__}: {exc}"

            checkpoints_saved[trainer.step] = {
                "checkpoint_dir": str(ckpt_dir),
                "state_finite": state_ok,
                "state_detail": state_detail,
                "reload_eval_success": eval_ok,
                "reload_eval_loss": eval_loss,
                "reload_eval_perplexity": eval_ppl,
                "reload_eval_detail": eval_detail,
            }
            if not state_ok or not eval_ok:
                stop_reason = (
                    f"checkpoint integrity/reload-evaluation failure at step "
                    f"{trainer.step}: state_ok={state_ok}, eval_ok={eval_ok}"
                )

        # Sustained-overfitting stop condition, copied verbatim from
        # training.experiments.scaling_scale3_budget_extension.run
        # (Directive: "reuse it rather than implementing a competing
        # definition"). Requires a consistent trend across 3
        # consecutive transitions in the last 4 recorded curve points,
        # not a single blip.
        if stop_reason is None and len(training_curve) >= 4:
            last4 = training_curve[-4:]
            val_rising = all(last4[i]["val_loss"] < last4[i + 1]["val_loss"] for i in range(3))
            train_falling = all(last4[i]["train_loss"] > last4[i + 1]["train_loss"] for i in range(3))
            if val_rising and train_falling:
                stop_reason = (
                    f"sustained overfitting detected: val loss rose and train "
                    f"loss fell across the last 4 recorded checkpoints "
                    f"(steps {[c['step'] for c in last4]})"
                )

    # Generation probes: ALL fixed prompts, unconditionally recorded
    # (Directive: "Do not select only successful-looking generations"),
    # run once at the end of training (or at the point training
    # stopped).
    for prompt in GENERATION_PROMPTS:
        prompt_ids = torch.tensor([encode_corpus(tokenizer, [prompt])], dtype=torch.long)
        generated_ids = generate(model, prompt_ids, max_new_tokens=10)
        generation_samples[prompt] = safe_decode_generation(tokenizer, generated_ids[0].tolist())

    final_train_loss_full_eval, final_val_loss = _current_losses()
    best_point = min(training_curve, key=lambda c: c["val_loss"])

    result.update(
        {
            "seed": SEED,
            "model_config": {
                "hidden_size": HIDDEN_SIZE,
                "num_layers": NUM_LAYERS,
                "num_heads": NUM_HEADS,
                "ffn_size": FFN_SIZE,
                "context_length": CONTEXT_LENGTH,
                "sequence_length": SEQUENCE_LENGTH,
                "vocab_size": model_config.vocab_size,
            },
            "parameter_count": parameter_count,
            "training_config": {
                "learning_rate": training_config.learning_rate,
                "batch_size": training_config.batch_size,
                "max_steps": training_config.max_steps,
                "grad_clip_norm": training_config.grad_clip_norm,
                "weight_decay": training_config.weight_decay,
            },
            "train_document_count": len(prep.train),
            "validation_document_count": len(prep.validation),
            "train_tokens": len(train_token_ids),
            "val_tokens": len(val_token_ids),
            "initial_train_loss": initial_train_loss,
            "final_train_loss_full_eval": final_train_loss_full_eval,
            "final_val_loss": final_val_loss,
            "best_val_loss": best_point["val_loss"],
            "best_val_step": best_point["step"],
            "uniform_baseline_loss": uniform_baseline_loss(model_config.vocab_size),
            "unigram_baseline_loss": unigram_baseline_loss(train_token_ids, val_token_ids),
            "training_curve": training_curve,
            "checkpoints_saved": checkpoints_saved,
            "generation_probes": generation_samples,
            "num_steps_completed": trainer.step,
            "reached_max_steps": trainer.step >= training_config.max_steps and stop_reason is None,

"""Tokenizer Vocabulary Scaling Investigation.

NOT a milestone: no NB-13 (or any other milestone) is assigned, and no
roadmap file is touched. This is a tokenizer-only experiment. It trains
NO Transformer.

QUESTION: how does tokenizer vocabulary size change tokenization
efficiency, correctness, determinism and vocabulary utilization on the
SAME fixed real corpus?

Reused, not duplicated:
  - acquisition: ``real_corpus_experiment.acquire_common_pile_wikimedia_sample``
  - corpus preparation / license gate / document-level split:
    ``training.corpus_prep.prepare_corpus`` with the same seed, val
    fraction and license policy as the first real-corpus experiment
  - tokenizer: the existing ``yug.tokenizer.ByteLevelBpeTokenizer``
    (unmodified); train-partition-only fitting goes through
    ``real_corpus_experiment.train_tokenizer_on_train_partition``
  - word-count definition: ``real_corpus_experiment._count_words``

FIT MODES
---------
``ByteLevelBpeTokenizer.train`` is a deterministic greedy algorithm
with first-seen tie-breaking, so the merge list learned for a target of
N tokens is a prefix of the merge list learned for any larger target on
the same ordered corpus. ``fit_mode="prefix"`` (default) exploits that:
train ONCE at the largest requested size, then assemble each smaller
tokenizer from the first K merges via the public constructor (special
tokens are re-assigned ids after the merges, exactly as ``train`` does).
This is a PROPOSAL-level efficiency shortcut, not an assumption: the
runner verifies it by also training independently (from scratch) for
every size <= ``verify_prefix_up_to`` and comparing tokenizer identity,
recording the outcome per size. ``fit_mode="independent"`` trains every
size from scratch and is the authoritative (slow) mode.

Either way, only TRAINING-partition text is ever passed to tokenizer
training. Validation text is used only for measurement afterwards.

METRIC DEFINITIONS (all documented, none invented silently)
-----------------------------------------------------------
- words: whitespace-split count (``_count_words``).
- tokens_per_word / tokens_per_character: total tokens / total words or
  characters over successfully encoded documents (same as
  ``compute_tokenizer_stats``).
- relative reduction vs baseline vocab B for metric m:
  ``(m_B - m_V) / m_B * 100``. Positive = fewer tokens than baseline.
- utilization universe: the non-special ids ``0 .. 256 + num_merges - 1``
  (256 base bytes + learned merges). Special tokens are reported
  separately because ``encode`` never emits them unless asked.
- observed: count > 0 in the encoded partition. never_observed:
  universe id with count 0.
- rare (threshold T): universe ids with count < T, INCLUDING never
  observed ones; thresholds are listed in the config.
- unigram entropy: Shannon entropy in bits of the empirical token
  distribution over the partition.
- ``observed_in_validation_not_in_train`` counts ids seen when encoding
  validation but with zero count when encoding train.

LIMITATIONS: see ``LIMITATIONS`` below; also reported in the result.
"""

def _pct_reduction(baseline: float | int | None, value: float | int | None) -> float | None:
    """``(baseline - value) / baseline * 100``; None if undefined."""
    if baseline is None or value is None or baseline == 0:
        return None
    return (baseline - value) / baseline * 100.0


def _nearest_rank(sorted_values: Sequence[int], q: float) -> int:
    """Nearest-rank percentile (deterministic, no interpolation)."""
    n = len(sorted_values)
    rank = max(1, math.ceil(q * n))
    return sorted_values[rank - 1]


def validate_vocab_sizes(vocab_sizes: Sequence[int], baseline: int, num_specials: int = len(SPECIAL_TOKENS)) -> list[int]:
    """Validate and return the sizes sorted ascending.

    Raises :class:`VocabScalingConfigError` for empty/duplicate/non-int
    sizes, sizes below 256 + specials, or a baseline not in the list.
    """
    sizes = list(vocab_sizes)
    if not sizes:
        raise VocabScalingConfigError("vocab_sizes must not be empty")
    for s in sizes:
        if isinstance(s, bool) or not isinstance(s, int):
            raise VocabScalingConfigError(f"vocab size must be int, got {s!r}")
    if len(set(sizes)) != len(sizes):
        raise VocabScalingConfigError(f"vocab_sizes contains duplicates: {sizes}")
    minimum = _NUM_BASE_BYTES + num_specials
    too_small = [s for s in sizes if s < minimum]
    if too_small:
        raise VocabScalingConfigError(f"vocab sizes {too_small} are below the minimum {minimum}")
    if baseline not in sizes:
        raise VocabScalingConfigError(f"baseline vocab size {baseline} must be one of vocab_sizes {sorted(sizes)}")
    return sorted(sizes)

def tokenizer_from_merge_prefix(full_tokenizer, target_vocab_size: int):
    """Assemble the tokenizer that independent training at
    ``target_vocab_size`` would produce, from the first K merges of a
    larger already-trained tokenizer (public constructor + public
    ``to_dict``; the tokenizer class is not modified).

    K = min(available merges, target - 256 - num_specials). Special
    tokens get ids after the merges, as ``ByteLevelBpeTokenizer.train``
    assigns them.
    """
    from yug.tokenizer.bpe import ByteLevelBpeTokenizer

    data = full_tokenizer.to_dict()
    names = sorted(data["special_tokens"], key=lambda n: data["special_tokens"][n])
    k = max(0, min(len(data["merges"]), target_vocab_size - _NUM_BASE_BYTES - len(names)))
    merges = [tuple(m) for m in data["merges"][:k]]
    specials = {name: _NUM_BASE_BYTES + k + i for i, name in enumerate(names)}
    return ByteLevelBpeTokenizer(merges=merges, special_tokens=specials)


def _train_independent(train_texts: list[str], vocab_size: int):
    from yug.tokenizer.bpe import ByteLevelBpeTokenizer

    return ByteLevelBpeTokenizer.train(corpus=train_texts, vocab_size=vocab_size, special_tokens=SPECIAL_TOKENS)

def measure_partition(tokenizer, texts: list[str]) -> tuple[dict[str, Any], Counter]:
    """Single encode pass over ``texts``: efficiency/correctness stats
    plus the token-id frequency counter.

    Stats keys and definitions match
    ``real_corpus_experiment.compute_tokenizer_stats`` (that function is
    verified equal on fixtures by a test), plus ``round_trip_mismatches``
    (``decode(encode(t)) != t``) and ``ids_out_of_range`` (emitted ids
    not in ``[0, vocab_size)`` or equal to a special-token id).
    """
    from yug.tokenizer.errors import DecodeError

    special_ids = set(tokenizer.special_tokens.values())
    vocab_size = tokenizer.vocab_size
    counts: Counter = Counter()
    chars = words = tokens = enc_fail = dec_fail = mismatches = out_of_range = 0

    for text in texts:
        chars += len(text)
        words += _count_words(text)
        try:
            ids = tokenizer.encode(text)
        except Exception:  # noqa: BLE001 - recorded, not fatal
            enc_fail += 1
            continue
        tokens += len(ids)
        counts.update(ids)
        try:
            decoded = tokenizer.decode(ids)
            if decoded != text:
                mismatches += 1
        except DecodeError:
            dec_fail += 1
    for tid in counts:
        if not (0 <= tid < vocab_size) or tid in special_ids:
            out_of_range += 1

    stats = {
        "document_count": len(texts),
        "total_characters": chars,
        "total_words": words,
        "total_tokens": tokens,
        "tokens_per_word": (tokens / words) if words else None,
        "tokens_per_character": (tokens / chars) if chars else None,
        "encoding_failures": enc_fail,
        "decoding_failures": dec_fail,
        "round_trip_mismatches": mismatches,
        "ids_out_of_range": out_of_range,
    }
    return stats, counts


def compute_utilization(
    counts: Counter,
    *,
    num_merges: int,
    num_specials: int,
    rare_thresholds: Sequence[int] = RARE_THRESHOLDS,
) -> dict[str, Any]:
    """Vocabulary-utilization summary for one partition's id counts.

    Universe = non-special ids ``0 .. 256 + num_merges - 1``. See module
    docstring for every definition.
    """
    universe_size = _NUM_BASE_BYTES + num_merges
    special_lo = universe_size
    special_ids = set(range(special_lo, special_lo + num_specials))

    in_universe = {tid: c for tid, c in counts.items() if 0 <= tid < universe_size and c > 0}
    observed = len(in_universe)
    never = [tid for tid in range(universe_size) if tid not in in_universe]
    total = sum(in_universe.values())

    freqs = sorted(in_universe.values())
    summary: dict[str, Any]
    if freqs:
        probs = [c / total for c in freqs]
        entropy = -sum(p * math.log2(p) for p in probs)
        desc = sorted(freqs, reverse=True)
        summary = {
            "min": freqs[0],
            "max": freqs[-1],
            "mean": total / observed,
            "median": _nearest_rank(freqs, 0.5),
            "p10": _nearest_rank(freqs, 0.10),
            "p90": _nearest_rank(freqs, 0.90),
            "p99": _nearest_rank(freqs, 0.99),
            "top10_token_share": sum(desc[:10]) / total,
            "top100_token_share": sum(desc[:100]) / total,
            "unigram_entropy_bits": entropy,
        }
    else:
        summary = {k: None for k in ("min", "max", "mean", "median", "p10", "p90", "p99", "top10_token_share", "top100_token_share", "unigram_entropy_bits")}

    never_base = sum(1 for t in never if t < _NUM_BASE_BYTES)
    return {
        "universe_size_non_special": universe_size,
        "tokens_observed": observed,
        "percent_observed": (observed / universe_size * 100.0) if universe_size else None,
        "tokens_never_observed": len(never),
        "never_observed_base_bytes": never_base,
        "never_observed_merges": len(never) - never_base,
        "never_observed_first_ids": never[:_MAX_LISTED_UNOBSERVED_IDS],
        "frequency_summary_observed_tokens": summary,
        "rare_token_counts": {
            f"count_lt_{t}": sum(1 for tid in range(universe_size) if in_universe.get(tid, 0) < t)
            for t in rare_thresholds
        },
        "special_tokens_emitted": sum(c for tid, c in counts.items() if tid in special_ids),
    }


def _fit_corpus_fingerprint(prep: CorpusPrepResult) -> str:
    import hashlib

    h = hashlib.sha256()
    for doc in prep.train:
        h.update(doc.content_hash.encode("utf-8"))
    return "sha256:" + h.hexdigest()

                == PREVIOUS_RUN_REFERENCE["accepted_content_hash"],
                "train_document_count_matches": len(prep.train) == PREVIOUS_RUN_REFERENCE["train_document_count"],
                "validation_document_count_matches": len(prep.validation)
                == PREVIOUS_RUN_REFERENCE["validation_document_count"],
            },
        },
        "leakage_checks": leakage,
        "tokenizers": [],
        "comparison_to_baseline": {},
        "errors": [],
        "limitations": list(LIMITATIONS),
        "timings_seconds_non_deterministic": {},
    }
    timings = result["timings_seconds_non_deterministic"]

    # --- fit -------------------------------------------------------------
    fitted: dict[int, Any] = {}
    fit_info: dict[int, dict[str, Any]] = {}
    try:
        if fit_mode == "prefix":
            t0 = time.perf_counter()
            say(f"fitting tokenizer once at max size {sizes[-1]} (train partition only)")
            full = train_tokenizer_on_train_partition(prep, vocab_size=sizes[-1])
            timings["fit_max_size"] = time.perf_counter() - t0
            for s in sizes:
                fitted[s] = tokenizer_from_merge_prefix(full, s)
                fit_info[s] = {"mode": "prefix_of_max_size_fit", "prefix_equals_independent_training": None}
    except Exception as exc:  # noqa: BLE001
        result["errors"].append({"stage": "fit_prefix", "error": f"{type(exc).__name__}: {exc}"})

        try:

                fit_mode == "prefix" and s in fitted and s <= verify_prefix_up_to
            )
            independent = None
            if need_independent:

                say(f"independent training at vocab {s}")
                independent = _train_independent(train_texts, s)
                timings[f"independent_fit_{s}"] = time.perf_counter() - t0
            if fit_mode == "independent":
                fitted[s] = independent
                fit_info[s] = {"mode": "independent", "prefix_equals_independent_training": None}
            elif independent is not None and s in fitted:
                fit_info[s]["prefix_equals_independent_training"] = independent.identity == fitted[s].identity
        except Exception as exc:  # noqa: BLE001
            result["errors"].append({"stage": "fit", "requested_vocab_size": s, "error": f"{type(exc).__name__}: {exc}"})

    smallest = sizes[0]
    if smallest in fitted:
        try:
            again = _train_independent(train_texts, smallest)
            result["determinism_check"] = {
                "requested_vocab_size": smallest,
                "retrain_identity_matches": again.identity == fitted[smallest].identity,
            }
        except Exception as exc:  # noqa: BLE001
            result["errors"].append({"stage": "determinism", "error": f"{type(exc).__name__}: {exc}"})

    # --- measure ---------------------------------------------------------
    for s in sizes:
        tok = fitted.get(s)
        if tok is None:
            continue
        try:
            t0 = time.perf_counter()
            say(f"measuring vocab {s} (actual {tok.vocab_size})")
            train_stats, train_counts = measure_partition(tok, train_texts)
            val_stats, val_counts = measure_partition(tok, val_texts)
            timings[f"measure_{s}"] = time.perf_counter() - t0

            num_specials = len(tok.special_tokens)
            num_merges = tok.vocab_size - _NUM_BASE_BYTES - num_specials
            util_train = compute_utilization(train_counts, num_merges=num_merges, num_specials=num_specials)
            util_val = compute_utilization(val_counts, num_merges=num_merges, num_specials=num_specials)
            util_val["observed_in_validation_not_in_train"] = sum(
                1 for tid, c in val_counts.items() if c > 0 and train_counts.get(tid, 0) == 0

                    "actual_vocab_size": tok.vocab_size,
                    "stopped_short_of_request": tok.vocab_size < s,
                    "num_merges": num_merges,
                    "tokenizer_identity": tok.identity,
                    "fit": fit_info.get(s, {}),
                    "train": train_stats,
                    "validation": val_stats,
                    "utilization": {"train": util_train, "validation": util_val},
                    "checks": {
                        "encoding_ok": train_stats["encoding_failures"] + val_stats["encoding_failures"] == 0,
                        "decoding_ok": train_stats["decoding_failures"] + val_stats["decoding_failures"] == 0,
                        "round_trip_ok": train_stats["round_trip_mismatches"] + val_stats["round_trip_mismatches"] == 0,

                        "no_special_or_out_of_range_ids_emitted": train_stats["ids_out_of_range"]
                        + val_stats["ids_out_of_range"]
                        == 0,
                        "save_load_identity_preserved": reloaded_identity == tok.identity,
                    },
                }
            )
        except Exception as exc:  # noqa: BLE001
            result["errors"].append({"stage": "measure", "requested_vocab_size": s, "error": f"{type(exc).__name__}: {exc}"})

    by_req = {t["requested_vocab_size"]: t for t in result["tokenizers"]}

    else:

                    base["validation"]["total_tokens"], t["validation"]["total_tokens"]
                ),

                    base["train"]["tokens_per_word"], t["train"]["tokens_per_word"]
                ),
                "validation_tokens_per_word_reduction_pct": _pct_reduction(
                    base["validation"]["tokens_per_word"], t["validation"]["tokens_per_word"]
                ),

                    base["train"]["tokens_per_character"], t["train"]["tokens_per_character"]
                ),
                "validation_tokens_per_character_reduction_pct": _pct_reduction(
                    base["validation"]["tokens_per_character"], t["validation"]["tokens_per_character"]
                ),
            }

                    == PREVIOUS_RUN_REFERENCE["validation_total_tokens"],
